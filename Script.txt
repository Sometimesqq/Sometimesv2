local function createESP(player)
    local drawings = {}
    
    drawings.Corners = {}
    for i = 1, 8 do
        local line = Drawing.new("Line")
        line.Visible = false
        line.Color = Config.ThemeColor
        line.Thickness = 1.5
        table.insert(drawings.Corners, line)
    end

    local healthBarBg = Drawing.new("Line")
    healthBarBg.Visible = false
    healthBarBg.Color = Color3.fromRGB(18, 20, 28)
    healthBarBg.Thickness = 3

    local healthBar = Drawing.new("Line")
    healthBar.Visible = false
    healthBar.Thickness = 1.5

    drawings.Skeleton = {}
    for i = 1, #R15Joints do
        local line = Drawing.new("Line")
        line.Visible = false
        line.Color = Config.ThemeColor
        line.Thickness = 1
        table.insert(drawings.Skeleton, line)
    end

    local text = Drawing.new("Text")
    text.Visible = false
    text.Size = 12
    text.Color = Color3.fromRGB(245, 248, 255)
    text.Center = true
    text.Outline = true

    espCache[player] = {
        Corners = drawings.Corners, 
        HealthBarBg = healthBarBg, 
        HealthBar = healthBar, 
        Skeleton = drawings.Skeleton,
        Text = text
    }
end

local function removeESP(player)
    if espCache[player] then
        for _, line in ipairs(espCache[player].Corners) do line:Remove() end
        espCache[player].HealthBarBg:Remove()
        espCache[player].HealthBar:Remove()
        for _, line in ipairs(espCache[player].Skeleton) do line:Remove() end
        espCache[player].Text:Remove()
        espCache[player] = nil
    end
end

for _, player in ipairs(Players:GetPlayers()) do
    if player ~= LocalPlayer then createESP(player) end
end

local function getTarget()
    if Config.SelectedTarget ~= "Closest" then
        local p = Players:FindFirstChild(Config.SelectedTarget)
        if p and p.Character and p.Character:FindFirstChild(Config.TargetPart) and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            return p
        end
        return nil
    end

    local closest = nil
    local shortestDist = Config.FOV
    local mousePos = UserInputService:GetMouseLocation()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            local part = player.Character:FindFirstChild(Config.TargetPart)
            if part then
                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closest = player
                    end
                end
            end
        end
    end
    return closest
end

local function updateThemeColors()

    if mainStrokeRef then mainStrokeRef.Color = Config.ThemeColor end
    if loaderGlowRef then loaderGlowRef.Color = Config.ThemeColor end
    if profileStrokeRef then profileStrokeRef.Color = Config.ThemeColor end
    
    for _, el in ipairs(themeFillElements) do

    end
end

-- Render Loop
local renderConnection = RunService.RenderStepped:Connect(function()
    FOVCircle.Visible = Config.ShowFOV
    if Config.ShowFOV then
        FOVCircle.Position = UserInputService:GetMouseLocation()
        FOVCircle.Radius = Config.FOV
        FOVCircle.Color = Config.ThemeColor
    end

    if Config.AimbotEnabled and Config.ESPEnabled and UserInputService:IsMouseButtonPressed(Config.AimbotKey) then
        local target = getTarget()
        if target and target.Character and target.Character:FindFirstChild(Config.TargetPart) then
            local targetPart = target.Character[Config.TargetPart]
            local aimPosition = targetPart.Position + Vector3.new(0, 1, 0)
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, aimPosition)
        end
    end

    for player, cache in pairs(espCache) do
        local char = player.Character
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")
        local humanoid = char and char:FindFirstChild("Humanoid")
        
        if Config.ESPEnabled and char and rootPart and humanoid and humanoid.Health > 0 then
            local vector, onScreen = Camera:WorldToViewportPoint(rootPart.Position)
            if onScreen then
                local headPos = Camera:WorldToViewportPoint(char.Head.Position + Vector3.new(0, 0.6, 0))
                local legPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, 3, 0))
                local height = math.abs(headPos.Y - legPos.Y)
                local width = height / 2

                local x = vector.X - width / 2
                local y = headPos.Y
                local w = width
                local h = height

                if Config.ShowBox then
                    local length = w / 4
                    cache.Corners[1].From = Vector2.new(x, y) cache.Corners[1].To = Vector2.new(x + length, y)
                    cache.Corners[2].From = Vector2.new(x, y) cache.Corners[2].To = Vector2.new(x, y + length)
                    cache.Corners[3].From = Vector2.new(x + w, y) cache.Corners[3].To = Vector2.new(x + w - length, y)
                    cache.Corners[4].From = Vector2.new(x + w, y) cache.Corners[4].To = Vector2.new(x + w, y + length)
                    cache.Corners[5].From = Vector2.new(x, y + h) cache.Corners[5].To = Vector2.new(x + length, y + h)
                    cache.Corners[6].From = Vector2.new(x, y + h) cache.Corners[6].To = Vector2.new(x, y + h - length)
                    cache.Corners[7].From = Vector2.new(x + w, y + h) cache.Corners[7].To = Vector2.new(x + w - length, y + h)
                    cache.Corners[8].From = Vector2.new(x + w, y + h) cache.Corners[8].To = Vector2.new(x + w, y + h - length)

                    for _, line in ipairs(cache.Corners) do
                        line.Color = Config.ThemeColor
                        line.Visible = true
                    end
                else
                    for _, line in ipairs(cache.Corners) do line.Visible = false end
                end

                if Config.ShowHealth then
                    local healthPercent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                    local barX = x - 6
                    
                    cache.HealthBarBg.From = Vector2.new(barX, y + h)
                    cache.HealthBarBg.To = Vector2.new(barX, y)
                    cache.HealthBarBg.Visible = true

                    cache.HealthBar.From = Vector2.new(barX, y + h)
                    cache.HealthBar.To = Vector2.new(barX, y + (h * (1 - healthPercent)))
                    cache.HealthBar.Color = Color3.fromRGB(255 - (healthPercent * 255), healthPercent * 255, 60)
                    cache.HealthBar.Visible = true
                else
                    cache.HealthBarBg.Visible = false
                    cache.HealthBar.Visible = false
                end

                if Config.ShowSkeleton then
                    for i, joint in ipairs(R15Joints) do
                        local p1 = char:FindFirstChild(joint[1])
                        local p2 = char:FindFirstChild(joint[2])
                        local line = cache.Skeleton[i]
                        
                        if p1 and p2 then
                            local v1, vis1 = Camera:WorldToViewportPoint(p1.Position)
                            local v2, vis2 = Camera:WorldToViewportPoint(p2.Position)
                            
                            if vis1 or vis2 then
                                line.From = Vector2.new(v1.X, v1.Y)
                                line.To = Vector2.new(v2.X, v2.Y)
                                line.Color = Config.ThemeColor
                                line.Visible = true
                            else
                                line.Visible = false
                            end
                        else
                            line.Visible = false
                        end
                    end
                else
                    for _, line in ipairs(cache.Skeleton) do line.Visible = false end
                end

                if Config.ShowInfo then
                    local distance = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and math.floor((rootPart.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude) or 0
                    cache.Text.Text = string.upper(player.Name) .. " [" .. math.floor(humanoid.Health) .. "HP | " .. distance .. "M]"
                    cache.Text.Position = Vector2.new(vector.X, y - 18)
                    cache.Text.Visible = true
                else
                    cache.Text.Visible = false
                end
            else
                for _, line in ipairs(cache.Corners) do line.Visible = false end
                cache.HealthBarBg.Visible = false
                cache.HealthBar.Visible = false
                for _, line in ipairs(cache.Skeleton) do line.Visible = false end
                cache.Text.Visible = false
            end
        else
            for _, line in ipairs(cache.Corners) do line.Visible = false end
            cache.HealthBarBg.Visible = false
            cache.HealthBar.Visible = false
            for _, line in ipairs(cache.Skeleton) do line.Visible = false end
            cache.Text.Visible = false
        end
    end
end)

local function UnloadHub()
    pcall(function() renderConnection:Disconnect() end)
    pcall(function() playerAddedConn:Disconnect() end)
    pcall(function() playerRemovingConn:Disconnect() end)
    pcall(function() FOVCircle:Remove() end)
    for _, cache in pairs(espCache) do
        for _, line in ipairs(cache.Corners) do pcall(function() line:Remove() end) end
        pcall(function() cache.HealthBarBg:Remove() end)
        pcall(function() cache.HealthBar:Remove() end)
        for _, line in ipairs(cache.Skeleton) do pcall(function() line:Remove() end) end
        pcall(function() cache.Text:Remove() end)
    end
    pcall(function() ScreenGui:Destroy() end)
end

local function AddToggle(parentTab, name, getState, callback)
    local f = Instance.new("Frame", parentTab)
    f.BackgroundColor3 = Config.CardBg
    f.BackgroundTransparency = 0.2
    f.Size = UDim2.new(1, 0, 0, 42)
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)

    local l = Instance.new("TextLabel", f)
    l.BackgroundTransparency = 1
    l.Position = UDim2.new(0, 14, 0, 0)
    l.Size = UDim2.new(0, 200, 1, 0)
    l.Font = Enum.Font.GothamBold
    l.Text = name
    l.TextColor3 = Color3.fromRGB(230, 235, 245)
    l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left

    local b = Instance.new("TextButton", f)
    local active = getState()
    b.BackgroundColor3 = active and Config.ThemeColor or Color3.fromRGB(28, 33, 46)
    b.Position = UDim2.new(1, -44, 0.5, -10)
    b.Size = UDim2.new(0, 32, 0, 20)
    b.Text = ""
    Instance.new("UICorner", b).CornerRadius = UDim.new(1, 0)

    if active then table.insert(themeFillElements, b) end

    b.MouseButton1Click:Connect(function()
        local curState = getState()
        curState = not curState
        callback(curState)
        
        if curState then
            table.insert(themeFillElements, b)
        else
            for i, el in ipairs(themeFillElements) do
                if el == b then table.remove(themeFillElements, i) break end
            end
        end

        TweenService:Create(b, TweenInfo.new(0.2), {
            BackgroundColor3 = curState and Config.ThemeColor or Color3.fromRGB(28, 33, 46)
        }):Play()
    end)
end

    listContainer.BackgroundColor3 = Color3.fromRGB(13, 16, 23)
    listContainer.BackgroundTransparency = 0.05
    listContainer.Position = UDim2.new(0, 0, 1, 4)
    listContainer.Size = UDim2.new(1, 0, 0, 0)
    listContainer.CanvasSize = UDim2.new(0, 0, 0, 0)
    listContainer.ScrollBarThickness = 2
    listContainer.Visible = false
    listContainer.ZIndex = 5
    Instance.new("UICorner", listContainer).CornerRadius = UDim.new(0, 8)

    local listLayout = Instance.new("UIListLayout", listContainer)
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Padding = UDim.new(0, 2)

    local isOpen = false

    local function updateDropdownList()
        for _, child in ipairs(listContainer:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end

        local options = {"Closest"}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                table.insert(options, p.Name)
            end
        end

        listContainer.Size = UDim2.new(1, 0, 0, math.min(#options * 28, 120))
        listContainer.CanvasSize = UDim2.new(0, 0, 0, #options * 28)

        for _, optName in ipairs(options) do
            local optBtn = Instance.new("TextButton", listContainer)
            optBtn.BackgroundColor3 = Color3.fromRGB(20, 24, 34)
            optBtn.BackgroundTransparency = 0.2
            optBtn.Size = UDim2.new(1, 0, 0, 26)
            optBtn.Font = Enum.Font.GothamBold
            optBtn.Text = optName
            optBtn.TextColor3 = Color3.fromRGB(210, 215, 230)
            optBtn.TextSize = 10
            optBtn.ZIndex = 6
            Instance.new("UICorner", optBtn).CornerRadius = UDim.new(0, 4)

            optBtn.MouseButton1Click:Connect(function()
                Config.SelectedTarget = optName
                dropdownBtn.Text = optName
                isOpen = false
                listContainer.Visible = false
            end)
        end
    end

    dropdownBtn.MouseButton1Click:Connect(function()
        isOpen = not isOpen
        if isOpen then
            updateDropdownList()
            listContainer.Visible = true
        else
            listContainer.Visible = false
        end
    end)
    
    UserInputService.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 and isOpen then
            listContainer.Visible = false
            isOpen = false
        end
    end)
end

-- Interactive Color Wheel Picker
local function AddInteractiveColorWheel(parentTab)
    local f = Instance.new("Frame", parentTab)
    f.BackgroundColor3 = Config.CardBg
    f.BackgroundTransparency = 0.2
    f.Size = UDim2.new(1, 0, 0, 210)
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)

    local l = Instance.new("TextLabel", f)
    l.BackgroundTransparency = 1
    l.Position = UDim2.new(0, 14, 0, 8)
    l.Size = UDim2.new(1, -28, 0, 16)
    l.Font = Enum.Font.GothamBold
    l.Text = "Custom Theme Wheel (Click & Drag)"
    l.TextColor3 = Color3.fromRGB(230, 235, 245)
    l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left

    local wheelHolder = Instance.new("TextButton", f)
    wheelHolder.BackgroundColor3 = Color3.fromRGB(13, 16, 23)
    wheelHolder.BackgroundTransparency = 0.5
    wheelHolder.Position = UDim2.new(0.5, -75, 0, 32)
    wheelHolder.Size = UDim2.new(0, 150, 0, 150)
    wheelHolder.Text = ""
    Instance.new("UICorner", wheelHolder).CornerRadius = UDim.new(1, 0)

    local resolution = 32
    local radius = 60
    local center = Vector2.new(75, 75)

    for i = 0, resolution do
        local angle = (i / resolution) * (math.pi * 2)
        local dot = Instance.new("Frame", wheelHolder)
        dot.Size = UDim2.new(0, 8, 0, 8)
        dot.AnchorPoint = Vector2.new(0.5, 0.5)
        dot.BackgroundColor3 = Color3.fromHSV(i / resolution, 1, 1)
        dot.BorderSizePixel = 0
        dot.Position = UDim2.new(0, center.X + math.cos(angle) * radius, 0, center.Y + math.sin(angle) * radius)
        Instance.new("UICorner", dot).CornerRadius = UDim.new(1, 0)
    end

    local draggingWheel = false
    wheelHolder.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then draggingWheel = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then draggingWheel = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if draggingWheel and input.UserInputType == Enum.UserInputType.MouseMovement then
            local absPos = wheelHolder.AbsolutePosition
            local mouseLoc = UserInputService:GetMouseLocation()
            local relX = mouseLoc.X - absPos.X - center.X
            local relY = mouseLoc.Y - absPos.Y - center.Y
            
            local angle = math.atan2(relY, relX)
            if angle < 0 then angle = angle + (math.pi * 2) end
            
            Config.ThemeColor = Color3.fromHSV(angle / (math.pi * 2), 1, 1)
            updateThemeColors()
        end
    end)
end
